# --- Sleep / Power Saving ---
# Enable auto-sleep after inactivity (true/false)
sleep_enabled = true

# Seconds of inactivity before sleeping
sleep_timeout = 120

# --- Audio ---
# Volume level (0-100)
volume = 80

# Playback speed as percentage (50=half speed, 100=normal, 150=fast)
# Slower speeds help pre-verbal users understand speech
playback_speed = 100

# --- Input ---
# Debounce time in seconds (prevents double-presses)
debounce_time = 0.5

# Flip encoder scroll direction (true/false)
encoder_direction_flip = false

# Play sound on button release instead of press (true/false)
# Helps prevent accidental activations. Emergency hold still
# triggers after timeout regardless of this setting.
play_on_release = false

# Show border around text on OLED (true/false)
# When false, displays 3 lines: previous, current, next item
show_border = false

# Show hint text overlay on graphic displays (true/false)
# Displays text_description at the bottom of the screen
display_hint_text = true

# Screen flipped 180° (variant default is 90)
display_rotation = 270

# Faster screen start-up (boot and every wake): ST7735R init with
# datasheet-style waits instead of the library's 500 ms ones.
st7735_fast_init = true

# Switch the amp's outputs off as soon as each sound finishes (no hiss
# between words), instead of holding the codec in reset after 15 s of quiet:
# the first word after a pause starts ~0.6 s sooner.
amp_soft_idle = true

# --- Hardware Variant ---
# Which build this physical device is. Overrides DEFAULT_VARIANT in
# hardware_config.py. Set this on any device that isn't the default:
# deploy.sh preserves an existing config.txt but overwrites
# hardware_config.py, so this is the setting that survives a deploy.
#
# Fruit Jam builds share one daughterboard socket — pick by fitted panel:
#   variant = FRUITJAM_V2       # 1.8" ST7735 (ZJY180SN00), 160x128, 3x2 grid
#   variant = FRUITJAM_LCD_28   # 2.8" ILI9341, 320x240, 4x2 grid
#
# An unknown name here logs a warning and falls back to DEFAULT_VARIANT.
# This board is the CLONE: TPS22917 display rail on D10, no I2C pull-ups yet,
# so audio is off. Switch to FRUITJAM_V2 once the pull-ups are fitted.
variant = FRUITJAM_CLONE_18

# --- Menu ---
# Starting menu file (in /menus directory)
start_menu = base_fruitjam.menu

# --- Emergency Push ---
# When enabled, holding the wake button during boot
# plays an emergency sound immediately.
emergency_push_enabled = true

# Sound file to play on emergency push
emergency_push_sound = /button_sounds/emergency.mp3

# Play emergency sound when button held for N seconds (true/false)
emergency_hold_enabled = true

# Seconds to hold before triggering emergency sound
emergency_hold_seconds = 3
